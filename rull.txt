| users | CREATE TABLE `users` (
  `id` int NOT NULL AUTO_INCREMENT,
  `username` varchar(50) NOT NULL,
  `password` varchar(100) NOT NULL,
  `role` enum('user','admin') NOT NULL DEFAULT 'user',
  `real_name` char(30) NOT NULL,
  `idcard` char(18) NOT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `username` (`username`),
  UNIQUE KEY `idcard` (`idcard`),
  UNIQUE KEY `idcard_2` (`idcard`)
) 


| orders | CREATE TABLE `orders` (
  `id` int NOT NULL AUTO_INCREMENT,
  `user_id` int NOT NULL,
  `train_id` int NOT NULL,
  `order_time` datetime DEFAULT CURRENT_TIMESTAMP,
  `status` enum('paid','cancelled','expired') DEFAULT NULL,
  `group_id` char(50) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `user_id` (`user_id`),
  KEY `train_id` (`train_id`),
  CONSTRAINT `orders_ibfk_1` FOREIGN KEY (`user_id`) REFERENCES `users` (`id`),
  CONSTRAINT `orders_ibfk_2` FOREIGN KEY (`train_id`) REFERENCES `trains` (`id`)
) 


| trains | CREATE TABLE `trains` (
  `id` int NOT NULL AUTO_INCREMENT,
  `train_number` varchar(20) NOT NULL,
  `departure` varchar(50) NOT NULL,
  `destination` varchar(50) NOT NULL,
  `departure_time` datetime NOT NULL,
  `total_seats` int NOT NULL,
  `available_seats` int NOT NULL,
  `price` decimal(7,2) NOT NULL DEFAULT '0.00',
  `is_pre_sal` enum('-1','0','1') DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `train_number` (`train_number`)
) 


| group_orders | CREATE TABLE `group_orders` (
  `group_id` varchar(50) NOT NULL,
  `applicant_user_id` int NOT NULL,
  `train_id` int NOT NULL,
  `total_passengers` int NOT NULL,
  `total_amount` decimal(10,2) NOT NULL,
  `status` enum('paid','cancelled','expired') NOT NULL DEFAULT 'cancelled',
  PRIMARY KEY (`group_id`)
) 


| group_passengers | CREATE TABLE `group_passengers` (
  `id` int NOT NULL AUTO_INCREMENT,
  `group_id` varchar(50) NOT NULL,

  PRIMARY KEY (`id`),
