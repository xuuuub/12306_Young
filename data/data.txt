--- Procedure: update_ispresal ---
BEGIN
    DECLARE current_time_val DATETIME;
    DECLARE fourteen_days_later DATE;
    DECLARE fifteen_days_later DATE;
    
    SET current_time_val = NOW();
    SET fourteen_days_later = DATE_ADD(DATE(current_time_val), INTERVAL 14 DAY);
    SET fifteen_days_later = DATE_ADD(DATE(current_time_val), INTERVAL 15 DAY);
    
    
    UPDATE trains 
    SET is_pre_sal = 
        CASE 
            
            WHEN departure_time > current_time_val 
                 AND departure_time < fifteen_days_later 
                 THEN '1'  
            
            
            WHEN DATE(departure_time) = fourteen_days_later 
                 THEN '0'  
            
            
            ELSE '-1'      
        END
    WHERE departure_time IS NOT NULL;
    
    
    SELECT CONCAT('成功更新了 ', ROW_COUNT(), ' 条记录的售票状态') as 执行结果;
    
END
-- End of Procedure --
